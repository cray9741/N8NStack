Gets cert for n8n/elastic
kubectl -n logging get secret elasticsearch-master-certs   -o jsonpath='{.data.ca\.crt}' | base64 -d > es-ca.crt

Getting Elastic User & Password:
ELASTIC_USER=$(kubectl -n logging get secret elasticsearch-master-credentials \
  -o jsonpath='{.data.username}' | base64 -d)

ELASTIC_PW=$(kubectl -n logging get secret elasticsearch-master-credentials \
  -o jsonpath='{.data.password}' | base64 -d)

echo "pw=$ELASTIC_USER"
echo "pw=$ELASTIC_PW"

Create gitub credentials
kubectl -n n8n create secret generic github-token   --from-literal=token=

Create AWS credentials
kubectl -n n8n create secret generic aws-credentials   --from-literal=token=
